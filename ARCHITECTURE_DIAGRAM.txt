/**

 * 
 * Navigation Layer (core/presentation/screens/navigation/)
 * ├── Routes.kt                  → Defines all route constants
 * ├── BottomNavItem.kt          → Sealed class for nav items

 * └── AppNavGraph.kt            → Main navigation setup
 * 
 * Screen Layer (core/presentation/screens/)
 * ├── HomeScreen.kt             → Product list (existing)
 * ├── CartScreen.kt             → Shopping cart (new)
 * ├── AccountScreen.kt          → User account (new)
 * └── component/
 *     └── DetailsProductScreen.kt → Product details (existing)
 * 
 * Theme Layer (ui/theme/)
 * ├── Theme.kt                  → Material3 theme
 * └── Color.kt                  → Color definitions
 */

/**
 * MATERIAL2 vs MATERIAL3 USAGE
 * 
 * Material2 Components (androidx.compose.material):
 * ├── Scaffold                   → Only for cutoutShape support
 * ├── BottomAppBar               → Only for cutoutShape support
 * ├── FloatingActionButton       → Compatible with Material2 Scaffold
 * ├── BottomNavigation           → Inside BottomAppBar
 * └── BottomNavigationItem       → Navigation items
 * 
 * Material3 Components (androidx.compose.material3):
 * ├── MaterialTheme              → Theme provider
 * │   ├── colorScheme           → Applied to Material2 components
 * │   └── typography            → Typography system
 * ├── Text                       → All text components
 * ├── Icon                       → Icons in screens
 * └── Everything else            → All other UI components
 * 
 * WHY THIS HYBRID?
 * Material3 doesn't support BottomAppBar.cutoutShape, which is required
 * for creating the notch/cutout for the FAB. So we use:
 * - Material2 for Scaffold + BottomAppBar (cutout functionality)
 * - Material3 colors applied to Material2 components (visual consistency)
 * - Material3 for everything else (modern design system)
 */

/**
 * NAVIGATION FLOW
 * 
 * User Interaction        Navigation Action           Destination
 * ─────────────────      ──────────────────          ────────────
 * Tap Home Icon    →     navController.navigate()  →  HomeScreen
 * Tap Account Icon →     navController.navigate()  →  AccountScreen
 * Tap FAB (Cart)   →     navController.navigate()  →  CartScreen
 * Tap Product      →     navController.navigate()  →  DetailsScreen
 * 
 * Navigation Stack Management:
 * - Uses NavController with single top flag
 * - Saves and restores state on navigation
 * - Pops to start destination to prevent deep stacks
 */

/**
 * KEY FEATURES
 * 
 * 1. Real Notch/Cutout
 *    - Uses Material2 BottomAppBar.cutoutShape = CircleShape
 *    - FAB is docked in center with isFloatingActionButtonDocked = true
 *    - Creates authentic Material Design cutout effect
 * 
 * 2. Clean Architecture
 *    - Separation of navigation, screens, and theme
 *    - Sealed classes for type-safe navigation items
 *    - Route constants for maintainability
 * 
 * 3. State Management
 *    - Navigation state persisted across configuration changes
 *    - Back stack properly managed
 *    - Selected state tracked via NavBackStackEntry
 * 
 * 4. Material3 Integration
 *    - Material3 theme colors applied throughout
 *    - Consistent visual language
 *    - Modern typography and design tokens
 */
